1//! Water rendering system with animated Gerstner waves
2//!
3//! Provides a chunked, camera-distance LOD ocean surface with:
4//! - 4 summed Gerstner wave components
5//! - Fresnel-based reflections
6//! - Depth-based color blending
7//! - Animated foam on wave crests
8//! - Discrete chunk-grid LOD (finer near the camera, coarser toward the horizon)
9//! - Per-chunk skirts that hide LOD-boundary cracks
10//! - Real water-level control via a shader uniform (no baked mesh Y)
11//!
12//! ## LOD / chunking model (W-series W.2a)
13//!
14//! The surface is a discrete grid of square chunks (`CHUNK_SIZE` units each)
15//! centered on the camera. Each frame, every chunk in the `(2*GRID_RADIUS+1)²`
16//! block around the camera picks an LOD by its distance, and chunks are drawn
17//! **instanced per LOD** — one indexed-instanced draw call per LOD level,
18//! independent of how many chunks chose that level. Wave displacement is a pure
19//! function of world XZ, so chunk meshes are world-stable (no swimming) and
20//! shared LOD-boundary vertices agree exactly; the only mismatch is curve-vs-chord
21//! along a coarse neighbour's edge (≤ wave amplitude), which the per-chunk skirt
22//! covers. This replaces the former single hardcoded `generate_water_plane(500,128)`
23//! plane.
24

30/// World-space edge length of one water chunk (units).
31const CHUNK_SIZE: f32 = 64.0;
32/// Chunk radius around the camera: a `(2*GRID_RADIUS+1)²` block is active.
33const GRID_RADIUS: i32 = 8;
34/// Maximum simultaneously-active chunks (instance-buffer capacity per LOD).
35const MAX_CHUNKS: usize = ((2 * GRID_RADIUS + 1) * (2 * GRID_RADIUS + 1)) as usize; // 289
36/// Per-LOD grid subdivision count (finest → coarsest).
37const LOD_SUBDIVS: [u32; 4] = [32, 16, 8, 4];
38/// Upper distance bound (world units, camera→chunk-center) for each LOD band.
39/// The last entry is the catch-all for everything farther out.
40const LOD_DISTANCES: [f32; 4] = [110.0, 220.0, 360.0, f32::INFINITY];
41/// How far skirt vertices drop below the displaced surface edge (units).
42/// Must exceed the maximum LOD-boundary height mismatch (≤ total wave
43/// amplitude ≈ 1.65 units) by a wide margin so no crack outruns the skirt.
44const SKIRT_DEPTH: f32 = 8.0;
45/// Default water level (world Y) — **THE world sea level**, sourced from
46/// [`astraweave_terrain::SEA_LEVEL`] (TW1 ratification #1: Y=2.0 is the single
47/// source of truth shared by the biome classifier's aquatic bands and the
48/// rendered water plane). Consumers that never call
49/// [`WaterRenderer::set_water_level`] get a surface at sea level.
50pub const DEFAULT_WATER_LEVEL: f32 = astraweave_terrain::SEA_LEVEL;

52// ── Horizon shell (T.W.1.A) ──────────────────────────────────────────────────
53//
54// The chunked LOD grid spans only ±(GRID_RADIUS + 0.5)·CHUNK_SIZE = ±544 world
55// units around the camera's chunk — at editor world scale its square edge was
56// exposed mid-water and tracked the camera (the T.W.1 render-gate rejection).
57// The horizon shell is a flat 8-quad annulus at water level extending the
58// surface from just inside the grid's outer boundary out past every camera far
59// plane in use, so the visible water boundary is the far-plane horizon arc (or
60// full fog), never a mesh edge. Chosen over (a) scaling GRID_RADIUS to the far
61// plane (radius ≈ 625 → ~1.56M chunks iterated per frame on the CPU) and
62// (b) far-LOD mega-tile rings (per-ring CPU + a NEW LOD-crack surface at every
63// ring boundary); the shell adds one draw of 16 triangles and no crack class —
64// the seam is overlap-covered and flat-by-construction (see WAVE_FADE_*).
65
66/// Inner half-extent of the horizon shell's hole, relative to the snapped
67/// camera-chunk center: one full chunk INSIDE the grid's outer boundary, so
68/// shell and grid overlap in a `CHUNK_SIZE`-wide band. Same-height,
69/// same-shader overlap is visually idempotent (the water fragment writes
70/// alpha = 1) and immune to the T-junction pinholes edge-abutment risks.
71const HORIZON_INNER_HALF_EXTENT: f32 = (GRID_RADIUS as f32 - 0.5) * CHUNK_SIZE; // 480
72/// Outer half-extent of the horizon shell. Must exceed the largest camera far
73/// plane the surface is judged under, so the shell's own edge is never inside
74/// the frustum. Regimes (`test_horizon_covers_both_view_regimes`): editor
75/// judging aids far=40,000 with fog 60k/120k (fog conceals NOTHING — it starts
76/// beyond the far plane); production far=5,000 with fog full at 1,800.
77pub const HORIZON_OUTER_HALF_EXTENT: f32 = 100_000.0;
78/// Camera-distance band over which Gerstner amplitude (and steepness, in the
79/// same ratio) fades to zero in the vertex shader. `WAVE_FADE_END` must leave
80/// the surface exactly FLAT before the shell's inner edge under the worst-case
81/// camera offset inside its chunk (32·√2 ≈ 45.3 → flat needed by ≈ 434.7), so
82/// the grid↔shell seam is flat-against-flat by construction — no new crack
83/// class. `WAVE_FADE_START` sits beyond the LOD0/LOD1 distance bands (≤ 220)
84/// so the near-field W-series look is untouched. MUST mirror the WGSL consts
85/// of the same names (`test_wgsl_mirrors_wave_fade_constants`).
86pub const WAVE_FADE_START: f32 = 260.0;
87pub const WAVE_FADE_END: f32 = 420.0;

89// ── Weave-response deformation (W.2c) ────────────────────────────────────────

91/// Maximum simultaneously-active weave-deformation instances (W.2c #4 ratification).
92/// Built to 8 — trivially raisable later, but the shader loops to exactly this bound
93/// so unused slots cost nothing beyond a `kind == None` early-continue.
94pub const MAX_WEAVE_INSTANCES: usize = 8;
95
96/// Peak world-space displacement a single weave can apply, in units. Equals
97/// [`SKIRT_DEPTH`] so a `Part`/`Raise` at full intensity can never outrun the LOD
98/// skirt and re-expose a seam (W.2c.2 skirt constraint). `intensity ∈ [0,1]` scales
99/// against this; the shader bounds each instance's height contribution to ±this.
100pub const WEAVE_MAX_DEFORM: f32 = SKIRT_DEPTH;
101
102/// Which weave-response deformation a [`WeaveInstance`] applies. Encoded as `u32`
103/// in the GPU instance; `None` (0) marks an inactive slot the shader skips.
104#[derive(Debug, Clone, Copy, PartialEq, Eq)]
105#[repr(u32)]
106pub enum WeaveKind {
107    /// Inactive slot — identity (no deformation).
108    None = 0,
109    /// Displace the surface *down*, exposing a corridor/well (additive negative height).
110    Part = 1,
111    /// Lift the surface *up* — a dome/ridge (additive positive height).
112    Raise = 2,
113    /// Lock/flatten the surface (suppress waves) + frozen material look. A mask, not
114    /// a displacement.
115    Freeze = 3,
116}
117
118/// A runtime weave-deformation instance: a normalized analytical profile placed at a
119/// world location. **Location lives only here.** The profile (selected by
120/// [`kind`](Self::kind)) is a position-agnostic shape in *local* space; the shader
121/// maps `world → local = rotate(world_xz − position, −orientation) / radius`,
122/// evaluates the profile, and scales by [`intensity`](Self::intensity). A single
123/// authored `Part` profile therefore serves a part-*anywhere* verb — only the
124/// instance carries where / how-big / which-way / how-strong.
125///
126/// Representation-agnostic (W.2c #2): a future deformation-texture profile would add a
127/// `profile_id` atlas index here without changing position/radius/orientation/
128/// intensity or the shader-feeding interface.
129#[derive(Debug, Clone, Copy)]
130pub struct WeaveInstance {
131    /// Which deformation this is (also selects the analytical profile in W.2c.2).
132    pub kind: WeaveKind,
133    /// World-space XZ center where the effect is placed (runtime input).
134    pub position: Vec2,
135    /// World-space footprint radius in units (runtime input): local `r = 1` maps here.
136    pub radius: f32,
137    /// Yaw orientation in radians (runtime input) — orients directional profiles.
138    pub orientation: f32,
139    /// Normalized magnitude in `[0, 1]` (runtime input). For `Part`/`Raise` it scales
140    /// the height against [`WEAVE_MAX_DEFORM`]; for `Freeze` it is the mask strength.
141    pub intensity: f32,
142    /// Animation phase / age in seconds (runtime input) — drives shader-side ramps.
143    pub phase: f32,
144}

146impl WeaveInstance {
147    fn to_raw(self) -> WeaveInstanceRaw {
148        WeaveInstanceRaw {
149            position: self.position.into(),
150            radius: self.radius.max(0.0),
151            orientation: self.orientation,
152            // Bound magnitude to [0,1]; with WEAVE_MAX_DEFORM == SKIRT_DEPTH in the
153            // shader this keeps any single deformation within skirt tolerance.
154            intensity: self.intensity.clamp(0.0, 1.0),
155            phase: self.phase,
156            kind: self.kind as u32,
157            _pad: 0,
158        }
159    }
160}

409impl WaterRenderer {
410    /// Create a new water renderer.
411    ///
412    /// `target_format` is the color format of the pass the water will draw
413    /// into — the **HDR target** (`Renderer::hdr_format()`, Rgba16Float), not
414    /// the window surface format. Installing a mismatched renderer is rejected
415    /// by `Renderer::set_water_renderer`.
416    pub fn new(
417        device: &wgpu::Device,
418        target_format: wgpu::TextureFormat,
419        depth_format: wgpu::TextureFormat,
420    ) -> Self {
421        // Load shader
422        let shader_source = include_str!("shaders/water.wgsl");
423        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
424            label: Some("water_shader"),
425            source: wgpu::ShaderSource::Wgsl(shader_source.into()),
426        });
427
428        // Uniform buffer
429        let uniforms = WaterUniforms::default();
430        let uniform_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
431            label: Some("water_uniforms"),
432            contents: bytemuck::bytes_of(&uniforms),
433            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
434        });
435
436        // Bind group layout — W.2b: 4 entries (uniform + scene-color + scene-depth
437        // + sampler) so the fragment shader can refract the scene behind the water
438        // and compute the depth-delta shoreline foam.
439        let bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
440            label: Some("water_bind_group_layout"),
441            entries: &[
442                wgpu::BindGroupLayoutEntry {
443                    binding: 0,
444                    visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
445                    ty: wgpu::BindingType::Buffer {
446                        ty: wgpu::BufferBindingType::Uniform,
447                        has_dynamic_offset: false,
448                        min_binding_size: None,
449                    },
450                    count: None,
451                },
452                // @1 scene color (opaque HDR snapshot) — sampled for refraction.
453                wgpu::BindGroupLayoutEntry {
454                    binding: 1,
455                    visibility: wgpu::ShaderStages::FRAGMENT,
456                    ty: wgpu::BindingType::Texture {
457                        sample_type: wgpu::TextureSampleType::Float { filterable: true },
458                        view_dimension: wgpu::TextureViewDimension::D2,
459                        multisampled: false,
460                    },
461                    count: None,
462                },
463                // @2 scene depth — read via textureLoad for the foam delta.
464                wgpu::BindGroupLayoutEntry {
465                    binding: 2,
466                    visibility: wgpu::ShaderStages::FRAGMENT,
467                    ty: wgpu::BindingType::Texture {
468                        sample_type: wgpu::TextureSampleType::Depth,
469                        view_dimension: wgpu::TextureViewDimension::D2,
470                        multisampled: false,
471                    },
472                    count: None,
473                },
474                // @3 sampler for the scene-color tap.
475                wgpu::BindGroupLayoutEntry {
476                    binding: 3,
477                    visibility: wgpu::ShaderStages::FRAGMENT,
478                    ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
479                    count: None,
480                },
481            ],
482        });
483
484        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
485            label: Some("water_scene_sampler"),
486            address_mode_u: wgpu::AddressMode::ClampToEdge,
487            address_mode_v: wgpu::AddressMode::ClampToEdge,
488            address_mode_w: wgpu::AddressMode::ClampToEdge,
489            mag_filter: wgpu::FilterMode::Linear,
490            min_filter: wgpu::FilterMode::Linear,
491            ..Default::default()
492        });
493
494        // 1×1 dummy scene textures so the initial bind group is valid before the
495        // renderer wires real scene-color/depth views via `prepare_scene`.
496        let dummy_scene_color = device.create_texture(&wgpu::TextureDescriptor {
497            label: Some("water_dummy_scene_color"),
498            size: wgpu::Extent3d {
499                width: 1,
500                height: 1,
501                depth_or_array_layers: 1,
502            },
503            mip_level_count: 1,
504            sample_count: 1,
505            dimension: wgpu::TextureDimension::D2,
506            format: wgpu::TextureFormat::Rgba16Float,
507            usage: wgpu::TextureUsages::TEXTURE_BINDING,
508            view_formats: &[],
509        });
510        let dummy_scene_color_view =
511            dummy_scene_color.create_view(&wgpu::TextureViewDescriptor::default());
512        let dummy_depth = device.create_texture(&wgpu::TextureDescriptor {
513            label: Some("water_dummy_depth"),
514            size: wgpu::Extent3d {
515                width: 1,
516                height: 1,
517                depth_or_array_layers: 1,
518            },
519            mip_level_count: 1,
520            sample_count: 1,
521            dimension: wgpu::TextureDimension::D2,
522            format: wgpu::TextureFormat::Depth32Float,
523            usage: wgpu::TextureUsages::TEXTURE_BINDING,
524            view_formats: &[],
525        });
526        let dummy_depth_view = dummy_depth.create_view(&wgpu::TextureViewDescriptor::default());
527
528        // Initial bind group references the dummy scene textures.
529        let bind_group = Self::build_bind_group(
530            device,
531            &bind_group_layout,
532            &uniform_buffer,
533            &dummy_scene_color_view,
534            &dummy_depth_view,
535            &sampler,
536        );
537
538        // Pipeline layout
539        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
540            label: Some("water_pipeline_layout"),
541            bind_group_layouts: &[&bind_group_layout],
542            push_constant_ranges: &[],
543        });
544
545        // Render pipeline with alpha blending.
546        // Two vertex buffers: per-vertex tile geometry + per-instance chunk offset.
547        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
548            label: Some("water_pipeline"),
549            layout: Some(&pipeline_layout),
550            vertex: wgpu::VertexState {
551                module: &shader,
552                entry_point: Some("vs_main"),
553                buffers: &[WaterVertex::desc(), ChunkInstance::desc()],
554                compilation_options: Default::default(),
555            },
556            fragment: Some(wgpu::FragmentState {
557                module: &shader,
558                entry_point: Some("fs_main"),
559                targets: &[Some(wgpu::ColorTargetState {
560                    format: target_format,
561                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
562                    write_mask: wgpu::ColorWrites::ALL,
563                })],
564                compilation_options: Default::default(),
565            }),
566            primitive: wgpu::PrimitiveState {
567                topology: wgpu::PrimitiveTopology::TriangleList,
568                strip_index_format: None,
569                front_face: wgpu::FrontFace::Ccw,
570                // Top surface (normal +Y) is CCW-front when viewed from above and
571                // skirt walls are wound outward-front, so back-face culling keeps
572                // the correct faces. The submerged/underside two-sided case is a
573                // deferred effects-phase concern (W-series Gemini triage §E).
574                cull_mode: Some(wgpu::Face::Back),
575                unclipped_depth: false,
576                polygon_mode: wgpu::PolygonMode::Fill,
577                conservative: false,
578            },
579            depth_stencil: Some(wgpu::DepthStencilState {
580                format: depth_format,
581                depth_write_enabled: false, // Transparent, don't write depth
582                depth_compare: wgpu::CompareFunction::LessEqual, // Normal depth testing
583                stencil: wgpu::StencilState::default(),
584                bias: wgpu::DepthBiasState::default(),
585            }),
586            multisample: wgpu::MultisampleState::default(),
587            multiview: None,
588            cache: None,
589        });

592        let mut lod_meshes = Vec::with_capacity(LOD_SUBDIVS.len());
593        let mut instance_buffers = Vec::with_capacity(LOD_SUBDIVS.len());
594        let mut instance_counts = Vec::with_capacity(LOD_SUBDIVS.len());
595        for (lod, &subdiv) in LOD_SUBDIVS.iter().enumerate() {

601            });
602            let index_buffer = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
603                label: Some(&format!("water_tile_ib_lod{lod}")),
604                contents: bytemuck::cast_slice(&indices),
605                usage: wgpu::BufferUsages::INDEX,
606            });
607            lod_meshes.push(LodMesh {
608                vertex_buffer,
609                index_buffer,
610                index_count: indices.len() as u32,
611            });
612            instance_buffers.push(device.create_buffer(&wgpu::BufferDescriptor {
613                label: Some(&format!("water_instances_lod{lod}")),
614                size: (MAX_CHUNKS * std::mem::size_of::<ChunkInstance>()) as wgpu::BufferAddress,
615                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
616                mapped_at_creation: false,
617            }));
618            instance_counts.push(0u32);
619        }
620
621        // Horizon shell (T.W.1.A): baked once; positioned per frame by a single
622        // ChunkInstance write in `update` so its overlapped inner edge always
623        // aligns with the grid's outer boundary.
624        let (shell_vertices, shell_indices) = Self::generate_horizon_shell();

626            vertex_buffer: device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
627                label: Some("water_horizon_vb"),
628                contents: bytemuck::cast_slice(&shell_vertices),
629                usage: wgpu::BufferUsages::VERTEX,
630            }),
631            index_buffer: device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
632                label: Some("water_horizon_ib"),
633                contents: bytemuck::cast_slice(&shell_indices),
634                usage: wgpu::BufferUsages::INDEX,
635            }),
636            index_count: shell_indices.len() as u32,
637        };
638        let horizon_instance_buffer = device.create_buffer(&wgpu::BufferDescriptor {
639            label: Some("water_horizon_instance"),
640            size: std::mem::size_of::<ChunkInstance>() as wgpu::BufferAddress,
641            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
642            mapped_at_creation: false,
643        });
644
645        Self {
646            pipeline,
647            target_format,
648            bind_group_layout,
649            bind_group,
650            uniform_buffer,
651            sampler,
652            _dummy_scene_color: dummy_scene_color,
653            _dummy_depth: dummy_depth,
654            scene_gen: u64::MAX,
655            lod_meshes,

658            horizon_mesh,
659            horizon_instance_buffer,
660            plane_visible: true,
661            volume_descs: Vec::new(),
662            volumes: Vec::new(),
663            uniforms,
664        }

707    /// Generate a subdivided water plane (surface grid only, local Y = 0,
708    /// centered at the origin). Reusable mesh primitive; [`Self::generate_tile`]
709    /// wraps it with a skirt.
710    fn generate_water_plane(size: f32, subdivisions: u32) -> (Vec<WaterVertex>, Vec<u32>) {
711        let mut vertices = Vec::new();
712        let mut indices = Vec::new();
713
714        let half_size = size / 2.0;
715        let step = size / subdivisions as f32;
716
717        // Generate vertices
718        for z in 0..=subdivisions {
719            for x in 0..=subdivisions {
720                let pos_x = -half_size + x as f32 * step;
721                let pos_z = -half_size + z as f32 * step;
722                let u = x as f32 / subdivisions as f32;
723                let v = z as f32 / subdivisions as f32;
724
725                vertices.push(WaterVertex {
726                    position: [pos_x, 0.0, pos_z], // local Y = 0; world Y from uniform
727                    uv: [u, v],
728                });
729            }
730        }
731
732        // Generate indices
733        for z in 0..subdivisions {
734            for x in 0..subdivisions {
735                let top_left = z * (subdivisions + 1) + x;
736                let top_right = top_left + 1;
737                let bottom_left = (z + 1) * (subdivisions + 1) + x;
738                let bottom_right = bottom_left + 1;
739
740                // First triangle
741                indices.push(top_left);
742                indices.push(bottom_left);
743                indices.push(top_right);
744
745                // Second triangle
746                indices.push(top_right);
747                indices.push(bottom_left);
748                indices.push(bottom_right);
749            }
750        }
751
752        (vertices, indices)
753    }

755    /// Append one outward-facing skirt wall segment between two adjacent top-edge
756    /// vertices. Bottom twins reuse the top XZ with sentinel local Y = -1.0. The
757    /// fixed winding `(a_top, b_top, a_bot)` + `(b_top, b_bot, a_bot)` yields an
758    /// outward horizontal normal when callers traverse each edge in the direction
759    /// documented in [`Self::generate_tile`].
760    fn push_wall(vertices: &mut Vec<WaterVertex>, indices: &mut Vec<u32>, a_top: u32, b_top: u32) {
761        let a = vertices[a_top as usize];
762        let b = vertices[b_top as usize];
763        let a_bot = vertices.len() as u32;
764        vertices.push(WaterVertex {
765            position: [a.position[0], -1.0, a.position[2]],
766            uv: a.uv,
767        });
768        let b_bot = vertices.len() as u32;
769        vertices.push(WaterVertex {
770            position: [b.position[0], -1.0, b.position[2]],
771            uv: b.uv,
772        });
773        indices.extend_from_slice(&[a_top, b_top, a_bot, b_top, b_bot, a_bot]);
774    }

776    /// Generate a chunk tile = surface grid + a perimeter skirt that hides
777    /// LOD-boundary cracks. Edges are traversed so the fixed skirt winding faces
778    /// outward (away from the chunk center): TOP +x, BOTTOM -x, LEFT -z, RIGHT +z.
779    fn generate_tile(size: f32, subdivisions: u32) -> (Vec<WaterVertex>, Vec<u32>) {
780        let (mut vertices, mut indices) = Self::generate_water_plane(size, subdivisions);
781        let n = subdivisions;
782        let stride = n + 1;
783
784        // TOP edge (z = 0 row), traverse x: 0 → n
785        for x in 0..n {
786            Self::push_wall(&mut vertices, &mut indices, x, x + 1);
787        }
788        // BOTTOM edge (z = n row), traverse x: n → 0
789        let zbase = n * stride;
790        for x in (1..=n).rev() {
791            Self::push_wall(&mut vertices, &mut indices, zbase + x, zbase + x - 1);
792        }
793        // LEFT edge (x = 0 col), traverse z: n → 0
794        for z in (1..=n).rev() {
795            Self::push_wall(&mut vertices, &mut indices, z * stride, (z - 1) * stride);
796        }
797        // RIGHT edge (x = n col), traverse z: 0 → n
798        for z in 0..n {
799            Self::push_wall(
800                &mut vertices,
801                &mut indices,
802                z * stride + n,
803                (z + 1) * stride + n,
804            );
805        }
806
807        (vertices, indices)
808    }

810    /// Generate the horizon shell: a flat square annulus in the grid's local
811    /// space, built as 8 quads (4 corner blocks + 4 edge strips) that share
812    /// complete edges — no T-junctions inside the shell. Surface-only (local
813    /// Y = 0, no skirt: the inner edge hides under the overlapping LOD grid,
814    /// the outer edge lies beyond every far plane). Triangles use the same
815    /// upward winding as [`Self::generate_water_plane`], asserted per-triangle
816    /// by `test_horizon_shell_geometry`.
817    fn generate_horizon_shell() -> (Vec<WaterVertex>, Vec<u32>) {
818        let b = [
819            -HORIZON_OUTER_HALF_EXTENT,
820            -HORIZON_INNER_HALF_EXTENT,
821            HORIZON_INNER_HALF_EXTENT,
822            HORIZON_OUTER_HALF_EXTENT,
823        ];
824        let mut vertices = Vec::with_capacity(32);
825        let mut indices = Vec::with_capacity(48);
826        for row in 0..3 {
827            for col in 0..3 {
828                if row == 1 && col == 1 {
829                    continue; // the hole the LOD grid fills (minus the overlap band)
830                }
831                let (x0, x1) = (b[col], b[col + 1]);
832                let (z0, z1) = (b[row], b[row + 1]);
833                let base = vertices.len() as u32;
834                for &(x, z) in &[(x0, z0), (x1, z0), (x0, z1), (x1, z1)] {
835                    vertices.push(WaterVertex {
836                        position: [x, 0.0, z],
837                        uv: [0.0, 0.0], // uv is unread by the fragment shader
838                    });
839                }
840                // Same (TL, BL, TR) / (TR, BL, BR) orientation as the grid tiles.
841                indices.extend_from_slice(&[
842                    base,
843                    base + 2,
844                    base + 1,
845                    base + 1,
846                    base + 2,
847                    base + 3,
848                ]);
849            }
850        }
851        (vertices, indices)
852    }

854    /// Generate a volume patch: a flat rectangular grid spanning
855    /// `[-hx, hx] × [-hz, hz]` in local space (surface-only, no skirt — a
856    /// volume's rim is either buried in the terrain bowl it fills or an
857    /// authored hard edge; residual displacement at calm styles is
858    /// sub-toleranace). Same winding family as [`Self::generate_water_plane`].
859    fn generate_volume_patch(half_extents: Vec2) -> (Vec<WaterVertex>, Vec<u32>) {
860        let n = VOLUME_SUBDIV;
861        let mut vertices = Vec::with_capacity(((n + 1) * (n + 1)) as usize);
862        let mut indices = Vec::with_capacity((n * n * 6) as usize);
863        for z in 0..=n {
864            for x in 0..=n {
865                let fx = (x as f32 / n as f32) * 2.0 - 1.0;
866                let fz = (z as f32 / n as f32) * 2.0 - 1.0;
867                vertices.push(WaterVertex {
868                    position: [fx * half_extents.x, 0.0, fz * half_extents.y],
869                    uv: [0.0, 0.0], // unread by the fragment shader
870                });
871            }
872        }
873        let stride = n + 1;
874        for z in 0..n {
875            for x in 0..n {
876                let tl = z * stride + x;
877                let tr = tl + 1;
878                let bl = (z + 1) * stride + x;
879                let br = bl + 1;
880                indices.extend_from_slice(&[tl, bl, tr, tr, bl, br]);
881            }
882        }
883        (vertices, indices)
884    }

990    /// Update water state for animation and recompute the per-LOD chunk set from
991    /// the camera position. Call once per frame before [`Self::render`].
992    pub fn update(&mut self, queue: &wgpu::Queue, view_proj: Mat4, camera_pos: Vec3, time: f32) {
993        self.uniforms.view_proj = view_proj.to_cols_array_2d();
994        // Inverse view-proj reconstructs the world position of the opaque scene
995        // behind the water from its sampled depth (W.2b depth-delta foam).
996        self.uniforms.inv_view_proj = view_proj.inverse().to_cols_array_2d();
997        self.uniforms.camera_pos = camera_pos.into();
998        self.uniforms.time = time;
999        queue.write_buffer(&self.uniform_buffer, 0, bytemuck::bytes_of(&self.uniforms));
1000
1001        // Assign the active chunk block around the camera to LOD bands.
1002        let cam_cx = (camera_pos.x / CHUNK_SIZE).floor() as i32;
1003        let cam_cz = (camera_pos.z / CHUNK_SIZE).floor() as i32;
1004        let mut per_lod: Vec<Vec<ChunkInstance>> =
1005            (0..LOD_SUBDIVS.len()).map(|_| Vec::new()).collect();
1006
1007        for dz in -GRID_RADIUS..=GRID_RADIUS {
1008            for dx in -GRID_RADIUS..=GRID_RADIUS {
1009                let cx = cam_cx + dx;
1010                let cz = cam_cz + dz;
1011                let center_x = (cx as f32 + 0.5) * CHUNK_SIZE;
1012                let center_z = (cz as f32 + 0.5) * CHUNK_SIZE;
1013                let dx_w = center_x - camera_pos.x;
1014                let dz_w = center_z - camera_pos.z;
1015                let dist = (dx_w * dx_w + dz_w * dz_w).sqrt();
1016                let lod = Self::lod_for_distance(dist);
1017                per_lod[lod].push(ChunkInstance {
1018                    offset: [center_x, center_z],
1019                });
1020            }
1021        }
1022
1023        for (lod, chunks) in per_lod.iter().enumerate() {
1024            self.instance_counts[lod] = chunks.len() as u32;
1025            if !chunks.is_empty() {
1026                queue.write_buffer(&self.instance_buffers[lod], 0, bytemuck::cast_slice(chunks));
1027            }
1028        }
1029
1030        // Anchor the horizon shell at the snapped camera-chunk center (the
1031        // grid's own anchor) so the overlap band always aligns exactly.
1032        let shell_anchor = ChunkInstance {
1033            offset: [
1034                (cam_cx as f32 + 0.5) * CHUNK_SIZE,
1035                (cam_cz as f32 + 0.5) * CHUNK_SIZE,
1036            ],
1037        };
1038        queue.write_buffer(
1039            &self.horizon_instance_buffer,
1040            0,
1041            bytemuck::bytes_of(&shell_anchor),
1042        );
1043
1044        // T.W.2: refresh each volume's uniform block with THIS frame's
1045        // view/camera/time — the global block with the volume's level/colors/
1046        // wave params, and NO weave instances (weaves are a global-plane
1047        // effect; an authored lake must not inherit a Part/Raise/Freeze aimed
1048        // at the sea). `screen_size` carries the last-prepared value (stale
1049        // for at most one frame across a resize, like the main block between
1050        // prepares).
1051        for (vol, desc) in self.volumes.iter().zip(&self.volume_descs) {
1052            let mut block = self.uniforms;
1053            block.water_level = desc.surface_level;
1054            block.water_color_deep = desc.color_deep;
1055            block.water_color_shallow = desc.color_shallow;
1056            block.foam_color = desc.color_foam;
1057            block.wave_amp_scale = desc.amplitude_scale.clamp(0.0, 1.0);
1058            block.foam_threshold = desc.foam_threshold;
1059            block.weave_count = 0;
1060            block.weave_instances = [WeaveInstanceRaw::ZERO; MAX_WEAVE_INSTANCES];
1061            queue.write_buffer(&vol.uniform_buffer, 0, bytemuck::bytes_of(&block));
1062        }
1063    }

1065    /// Wire the scene-color snapshot + scene-depth views into the water bind group
1066    /// and upload the screen size, immediately before the water pass (W.2b). The
1067    /// bind group is only rebuilt when `resource_gen` changes (resize); the uniform
1068    /// upload is cheap and unconditional. Must be called each frame before
1069    /// [`Self::render`] in the split water pass.
1070    pub fn prepare_scene(
1071        &mut self,
1072        device: &wgpu::Device,
1073        queue: &wgpu::Queue,
1074        scene_color_view: &wgpu::TextureView,
1075        scene_depth_view: &wgpu::TextureView,
1076        screen_size: [f32; 2],
1077        resource_gen: u64,
1078    ) {
1079        if self.scene_gen != resource_gen {
1080            self.bind_group = Self::build_bind_group(
1081                device,
1082                &self.bind_group_layout,
1083                &self.uniform_buffer,
1084                scene_color_view,
1085                scene_depth_view,
1086                &self.sampler,
1087            );
1088            // T.W.2: volume bind groups share the invalidation (also forced

1222    /// Render the water surface: the global plane (one instanced draw per
1223    /// active LOD band + the horizon shell) when `plane_visible`, then any
1224    /// authored volume patches (T.W.2 — each with its own uniform block).
1225    pub fn render<'a>(&'a self, render_pass: &mut wgpu::RenderPass<'a>) {
1226        render_pass.set_pipeline(&self.pipeline);
1227
1228        if self.plane_visible {
1229            render_pass.set_bind_group(0, &self.bind_group, &[]);
1230            for (lod, mesh) in self.lod_meshes.iter().enumerate() {
1231                let count = self.instance_counts[lod];
1232                if count == 0 {
1233                    continue;
1234                }
1235                render_pass.set_vertex_buffer(0, mesh.vertex_buffer.slice(..));
1236                render_pass.set_vertex_buffer(1, self.instance_buffers[lod].slice(..));
1237                render_pass
1238                    .set_index_buffer(mesh.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1239                render_pass.draw_indexed(0..mesh.index_count, 0, 0..count);
1240            }
1241
1242            // Horizon shell (T.W.1.A): extends the surface past every far
1243            // plane. Guarded by the same dormant condition as the grid, so
1244            // no visible chunks still means "the plane draws nothing".
1245            if self.has_visible_chunks() {
1246                render_pass.set_vertex_buffer(0, self.horizon_mesh.vertex_buffer.slice(..));
1247                render_pass.set_vertex_buffer(1, self.horizon_instance_buffer.slice(..));
1248                render_pass.set_index_buffer(
1249                    self.horizon_mesh.index_buffer.slice(..),
1250                    wgpu::IndexFormat::Uint32,
1251                );
1252                render_pass.draw_indexed(0..self.horizon_mesh.index_count, 0, 0..1);
1253            }
1254        }
1255
1256        // T.W.2 volume patches — drawn regardless of the plane flag.
1257        for vol in &self.volumes {
1258            render_pass.set_bind_group(0, &vol.bind_group, &[]);
1259            render_pass.set_vertex_buffer(0, vol.vertex_buffer.slice(..));
1260            render_pass.set_vertex_buffer(1, vol.instance_buffer.slice(..));
1261            render_pass.set_index_buffer(vol.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
1262            render_pass.draw_indexed(0..vol.index_count, 0, 0..1);
1263        }
1264    }
