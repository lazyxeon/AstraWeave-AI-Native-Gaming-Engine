4// Canonical vertex layout for engine meshes: P/N/T/UV
5#[repr(C)]
6#[derive(Copy, Clone, Debug, Pod, Zeroable)]
7pub struct MeshVertex {
8    pub position: [f32; 3],
9    pub normal: [f32; 3],
10    pub tangent: [f32; 4], // xyz=tangent, w=handedness
11    pub uv: [f32; 2],
12}
13
14impl MeshVertex {
15    pub fn new(position: Vec3, normal: Vec3, tangent: Vec4, uv: Vec2) -> Self {
16        Self {
17            position: position.to_array(),
18            normal: normal.to_array(),
19            tangent: tangent.to_array(),
20            uv: uv.to_array(),
21        }
22    }
23    pub fn from_arrays(
24        position: [f32; 3],
25        normal: [f32; 3],
26        tangent: [f32; 4],
27        uv: [f32; 2],
28    ) -> Self {
29        Self {
30            position,
31            normal,
32            tangent,
33            uv,
34        }
35    }
36    pub const ATTRIBS: [wgpu::VertexAttribute; 4] = wgpu::vertex_attr_array![
37        0 => Float32x3, // position
38        1 => Float32x3, // normal
39        2 => Float32x4, // tangent
40        3 => Float32x2, // uv
41    ];
42}

76impl CpuMesh {
77    pub fn aabb(&self) -> Option<(Vec3, Vec3)> {
78        if self.vertices.is_empty() {
79            return None;
80        }
81        let mut min = Vec3::splat(f32::INFINITY);
82        let mut max = Vec3::splat(f32::NEG_INFINITY);
83        for v in &self.vertices {
84            let p = Vec3::from_array(v.position);
85            min = min.min(p);
86            max = max.max(p);
87        }
88        Some((min, max))
89    }
90}

92// MikkTSpace tangent generation (spec-compliant)
93//
94// Uses the `mikktspace` crate which implements the exact algorithm from
95// Morten Mikkelsen's paper. This guarantees tangent-space consistency
96// with normal maps authored in MikkTSpace-aware tools (Blender, Substance, xNormal).
97
98/// Adapter for `mikktspace::Geometry` over indexed triangle meshes.
99struct MikkTSpaceAdapter<'a> {
100    vertices: &'a mut [MeshVertex],
101    indices: &'a [u32],
102}
103
104impl mikktspace::Geometry for MikkTSpaceAdapter<'_> {
105    fn num_faces(&self) -> usize {
106        self.indices.len() / 3
107    }
108
109    fn num_vertices_of_face(&self, _face: usize) -> usize {
110        3
111    }
112
113    fn position(&self, face: usize, vert: usize) -> [f32; 3] {
114        let idx = self.indices[face * 3 + vert] as usize;
115        self.vertices[idx].position
116    }
117
118    fn normal(&self, face: usize, vert: usize) -> [f32; 3] {
119        let idx = self.indices[face * 3 + vert] as usize;
120        self.vertices[idx].normal
121    }
122
123    fn tex_coord(&self, face: usize, vert: usize) -> [f32; 2] {
124        let idx = self.indices[face * 3 + vert] as usize;
125        self.vertices[idx].uv
126    }
127
128    fn set_tangent_encoded(&mut self, tangent: [f32; 4], face: usize, vert: usize) {
129        let idx = self.indices[face * 3 + vert] as usize;
130        self.vertices[idx].tangent = tangent;
131    }
132}

134/// Compute MikkTSpace-compliant tangents for an indexed triangle mesh.
135///
136/// Falls back to a Lengyel approximation if the mesh has degenerate
137/// geometry that MikkTSpace cannot handle (e.g., zero-area triangles).
138pub fn compute_tangents(mesh: &mut CpuMesh) {
139    if mesh.indices.len() % 3 != 0 || mesh.indices.is_empty() {
140        return;
141    }
142
143    // MikkTSpace panics on degenerate triangles where all three indices
144    // reference the same vertex (causes out-of-bounds in the algorithm).
145    // Check for this and fall back to Lengyel if found.
146    let has_degenerate = mesh
147        .indices
148        .chunks_exact(3)
149        .any(|tri| tri[0] == tri[1] || tri[1] == tri[2] || tri[0] == tri[2]);
150    if has_degenerate {
151        compute_tangents_lengyel(mesh);
152        return;
153    }
154
155    // MikkTSpace also panics when the mesh collapses to zero spatial extent —
156    // every referenced vertex position coincident, even via DISTINCT indices
157    // (which the index check above misses). Its vertex-weld spatial hash
158    // (mikktspace-0.3.0 generated.rs:1597) divides by the bounding-box extent in
159    // FindGridCell, so a zero-extent box yields 0/0 = NaN, saturating every vertex
160    // into hash cell 0 and overflowing the per-cell offset table — in the hashing
161    // pre-pass, *before* MikkTSpace's own per-triangle degeneracy skip can run.
162    // A mesh with any positional spread keeps a non-degenerate box and is handled
163    // correctly (MikkTSpace's internal skip drops individual zero-area triangles),
164    // so we reroute only when EVERY triangle is geometrically degenerate. The
165    // Lengyel fallback is finite-by-construction (`.max(1e-8)` UV-determinant clamp
166    // + `normalize_or_zero`), so it yields finite tangents for this input. A
167    // non-finite position would likewise poison FindGridCell, so treat it as
168    // degenerate too.
169    let all_zero_area = mesh.indices.chunks_exact(3).all(|tri| {
170        let p0 = Vec3::from_array(mesh.vertices[tri[0] as usize].position);
171        let p1 = Vec3::from_array(mesh.vertices[tri[1] as usize].position);
172        let p2 = Vec3::from_array(mesh.vertices[tri[2] as usize].position);
173        let area_sq = (p1 - p0).cross(p2 - p0).length_squared();
174        !area_sq.is_finite() || area_sq <= 1e-20
175    });
176    if all_zero_area {
177        compute_tangents_lengyel(mesh);
178        return;
179    }
180
181    let mut adapter = MikkTSpaceAdapter {
182        vertices: &mut mesh.vertices,
183        indices: &mesh.indices,
184    };
185
186    if !mikktspace::generate_tangents(&mut adapter) {
187        // MikkTSpace can fail on degenerate geometry — fall back to
188        // Lengyel approximation so we never leave tangents unset.
189        compute_tangents_lengyel(mesh);
190    }
191}

193/// Lengyel approximation fallback (Gram-Schmidt orthonormalization + handedness).
194fn compute_tangents_lengyel(mesh: &mut CpuMesh) {
195    let v = &mut mesh.vertices;
196    let idx = &mesh.indices;
197    let mut tan1: Vec<Vec3> = vec![Vec3::ZERO; v.len()];
198    let mut tan2: Vec<Vec3> = vec![Vec3::ZERO; v.len()];
199    for tri in idx.chunks_exact(3) {
200        let (i0, i1, i2) = (tri[0] as usize, tri[1] as usize, tri[2] as usize);
201        let p0 = Vec3::from_array(v[i0].position);
202        let p1 = Vec3::from_array(v[i1].position);
203        let p2 = Vec3::from_array(v[i2].position);
204        let uv0 = Vec2::from_array(v[i0].uv);
205        let uv1 = Vec2::from_array(v[i1].uv);
206        let uv2 = Vec2::from_array(v[i2].uv);
207        let dp1 = p1 - p0;
208        let dp2 = p2 - p0;
209        let duv1 = uv1 - uv0;
210        let duv2 = uv2 - uv0;
211        let r = 1.0 / (duv1.x * duv2.y - duv1.y * duv2.x).max(1e-8);
212        let sdir = (dp1 * duv2.y - dp2 * duv1.y) * r;
213        let tdir = (dp2 * duv1.x - dp1 * duv2.x) * r;
214        tan1[i0] += sdir;
215        tan1[i1] += sdir;
216        tan1[i2] += sdir;
217        tan2[i0] += tdir;
218        tan2[i1] += tdir;
219        tan2[i2] += tdir;
220    }
221    for i in 0..v.len() {
222        let n = Vec3::from_array(v[i].normal).normalize_or_zero();
223        let t = tan1[i];
224        let tangent = (t - n * n.dot(t)).normalize_or_zero();
225        let w = if n.cross(t).dot(tan2[i]) < 0.0 {
226            -1.0
227        } else {
228            1.0
229        };
230        v[i].tangent = [tangent.x, tangent.y, tangent.z, w];
231    }
232}
